# Compiles every public header the way a consumer of the -dev package compiles
# it, and fails the build if that is not possible.
#
# Nothing else in this project ever does that. The in-tree build puts all of
# src/ on the include path (include_directories(${INCLUDE_ROOT}) in the root
# CMakeLists.txt) and defines SLIDEIO_INTERNAL_HEADER globally, so a public
# header that includes an internal one, or that only compiles because some
# other header happened to be included first, builds green in-tree and breaks
# only for somebody who downloaded a package. auxfiles/package-smoke catches
# the same class of problem, but only for what its main.cpp reaches: it includes
# slideio.hpp, whose include closure is 8 of the 37 public headers. The other 29
# have never been compiled by a consumer, here or in CI.
#
# Two things make this directory different from every other one, and both are
# load-bearing:
#
#   - the include path is the staged public tree and nothing else, so an
#     include of an internal header cannot resolve;
#   - SLIDEIO_INTERNAL_HEADER is not defined, which is what a consumer's
#     compiler does. rect.hpp, size.hpp and range.hpp are guarded on it and
#     pull <opencv2/core/types.hpp> when it is set, so compiling them with it
#     defined would test the opposite of the interesting case.
#
# Both come from clearing the inherited directory properties below. Remove
# either and the check still passes -- vacuously, and forever.
set_property(DIRECTORY PROPERTY INCLUDE_DIRECTORIES "")
set_property(DIRECTORY PROPERTY COMPILE_DEFINITIONS "")

set(PUBLIC_HEADER_STAGE ${CMAKE_BINARY_DIR}/public-include)

# Staged with configure_file rather than file(COPY) on purpose. configure_file
# registers its input as a configure dependency, so editing a public header
# re-runs cmake and re-stages it. file(COPY) does not, which is the trap
# documented in CLAUDE.md for extern/pole: the build compiles the stale copy,
# silently.
set(_generated_sources)
set(_expected_staged)

foreach(_module IN LISTS SLIDEIO_PUBLIC_HEADER_MODULES)
    foreach(_name IN LISTS SLIDEIO_PUBLIC_HEADERS_${_module})
        set(_source ${INCLUDE_ROOT}/slideio/${_module}/${_name})
        if(NOT EXISTS ${_source})
            message(FATAL_ERROR
                "public-headers.cmake for module '${_module}' lists ${_name}, "
                "which does not exist at ${_source}.")
        endif()
        configure_file(${_source} ${PUBLIC_HEADER_STAGE}/slideio/${_module}/${_name} COPYONLY)

        # One translation unit per header, including it first and alone: that is
        # what makes the header prove it is self-contained rather than riding on
        # whatever a previous include dragged in. The second include is not a
        # copy-paste slip -- it checks the include guard, without which a
        # consumer including two of our headers gets redefinition errors.
        string(MAKE_C_IDENTIFIER "${_module}_${_name}" _id)
        set(_tu ${CMAKE_CURRENT_BINARY_DIR}/tu/${_id}.cpp)
        file(GENERATE OUTPUT ${_tu} CONTENT
"// Generated by src/public_headers_check/CMakeLists.txt. Do not edit.
#include <slideio/${_module}/${_name}>
#include <slideio/${_module}/${_name}>
extern const int ${_id}_compiled;
const int ${_id}_compiled = 0;
")
        list(APPEND _generated_sources ${_tu})
        list(APPEND _expected_staged slideio/${_module}/${_name})
    endforeach()
endforeach()

list(LENGTH _generated_sources _header_count)
message(STATUS "Public header check: ${_header_count} headers")

# OBJECT: the objects are never linked or installed. Compiling is the whole test.
add_library(slideio_public_headers_check OBJECT ${_generated_sources})
target_include_directories(slideio_public_headers_check PRIVATE ${PUBLIC_HEADER_STAGE})
set_target_properties(slideio_public_headers_check PROPERTIES FOLDER "checks")

# configure_file copies, it never removes. A header dropped from a
# public-headers.cmake therefore leaves its staged copy behind, where it goes on
# compiling and goes on being found -- so the check would keep certifying a
# header that is no longer public, and would keep letting other public headers
# include it. Compare the staged tree against what the lists actually name.
#
# An earlier version of this tried to prove the point by try_compile-ing an
# internal header against the staged tree and asserting the compile failed. That
# does not work, twice over: cvscene.hpp includes <opencv2/core.hpp>, so the
# probe failed for a reason that had nothing to do with staging and would have
# reported success even while the header sat in the tree; and try_compile caches
# its result variable, so it would have run once in the life of a build tree.
file(GLOB_RECURSE _staged_files RELATIVE ${PUBLIC_HEADER_STAGE} ${PUBLIC_HEADER_STAGE}/*)
set(_stale)
foreach(_staged IN LISTS _staged_files)
    if(NOT _staged IN_LIST _expected_staged)
        list(APPEND _stale ${_staged})
    endif()
endforeach()
if(_stale)
    string(REPLACE ";" ", " _stale_text "${_stale}")
    message(FATAL_ERROR
        "The staged public header tree at ${PUBLIC_HEADER_STAGE} holds files "
        "that no public-headers.cmake names: ${_stale_text}. They are almost "
        "certainly headers that were public in an earlier configure of this "
        "build tree. Delete ${PUBLIC_HEADER_STAGE} and re-configure.")
endif()
